<script>
    /**
     * 1. 백엔드(FastAPI) 데이터 비동기 호출 및 화면 렌더링 로직
     */
    async function fetchAndRenderNotices() {
        const apiUrl = 'http://127.0.0.1:8000/api/notices'; 
        const tbody = document.getElementById('notice-list-body');

        try {
            const response = await fetch(apiUrl);
            
            if (!response.ok) {
                throw new Error('HTTP 통신 에러 발생: ' + response.status);
            }

            const data = await response.json();
            tbody.innerHTML = '';

            if (data.length === 0) {
                tbody.innerHTML = '<tr><td colspan="4" style="text-align: center;">등록된 창업 지원 공고가 없습니다.</td></tr>';
                return;
            }

            data.forEach(notice => {
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td style="text-align: center; font-weight: bold; color: #555;">${notice.agency_name}</td>
                    <td>
                        <a href="${notice.link}" target="_blank" rel="noopener noreferrer" style="text-decoration: none; color: #333; font-weight: bold;">
                            ${notice.title}
                        </a>
                    </td>
                    <td style="text-align: center;">${notice.reg_date}</td>
                    <td style="text-align: center;">
                        <a href="${notice.link}" target="_blank" rel="noopener noreferrer" class="btn-link">상세보기</a>
                    </td>
                `;
                tbody.appendChild(tr);
            });

        } catch (error) {
            console.warn('백엔드 서버 미연결 상태. UI 테스트용 모의 데이터를 렌더링함.', error);
            // 백엔드가 꺼져있을 때 검색 및 UI 테스트를 위한 가상 데이터 표출
            tbody.innerHTML = `
                <tr>
                    <td style="text-align: center; font-weight: bold; color: #555;">경북테크노파크</td>
                    <td><a href="#" style="text-decoration: none; color: #333; font-weight: bold;">2026년 경북 이차전지 리사이클링 상용화 기술 지원 공고</a></td>
                    <td style="text-align: center;">2026-11-17</td>
                    <td style="text-align: center;"><a href="#" class="btn-link">상세보기</a></td>
                </tr>
                <tr>
                    <td style="text-align: center; font-weight: bold; color: #555;">경북창조경제혁신센터</td>
                    <td><a href="#" style="text-decoration: none; color: #333; font-weight: bold;">친환경 설비 분야 초기창업패키지 모집 안내</a></td>
                    <td style="text-align: center;">2026-11-16</td>
                    <td style="text-align: center;"><a href="#" class="btn-link">상세보기</a></td>
                </tr>
                <tr>
                    <td style="text-align: center; font-weight: bold; color: #555;">기술보증기금</td>
                    <td><a href="#" style="text-decoration: none; color: #333; font-weight: bold;">벤처기업 혁신성장 지원 특별보증 (최대 100억)</a></td>
                    <td style="text-align: center;">2026-11-15</td>
                    <td style="text-align: center;"><a href="#" class="btn-link">상세보기</a></td>
                </tr>
            `;
        }
    }

    /**
     * 2. 실시간 키워드 필터링 로직 (검색창 입력 이벤트)
     */
    document.getElementById('noticeSearchInput').addEventListener('input', function(event) {
        const keyword = event.target.value.toLowerCase().trim();
        const tbody = document.getElementById('notice-list-body');
        const rows = tbody.getElementsByTagName('tr');

        for (let i = 0; i < rows.length; i++) {
            const row = rows[i];
            
            // 데이터가 없는 기본 안내 셀 등은 검사에서 제외
            if (row.cells.length < 4) continue;

            const agencyName = row.cells[0].textContent.toLowerCase();
            const titleText = row.cells[1].textContent.toLowerCase();

            // 기관명이나 공고 제목에 키워드가 포함되어 있으면 표출, 아니면 숨김
            if (agencyName.includes(keyword) || titleText.includes(keyword)) {
                row.style.display = '';
            } else {
                row.style.display = 'none';
            }
        }
    });

    // 브라우저 렌더링 직후 데이터 로드 실행
    document.addEventListener('DOMContentLoaded', fetchAndRenderNotices);
</script>
